   // -------------------------------------------------------------------------
   // Main 2D Render Pipeline Dispatcher
   // -------------------------------------------------------------------------
   @EventHandler
   public void onRender2D(Render2DEvent event) {
      if (this.mc == null || this.mc.field_1724 == null || this.mc.field_1687 == null) {
         return;
      }

      class_332 context = event.getDrawContext();
      if (context == null) {
         return;
      }

      try {
         int sw = RenderUtil.getScaledWidth();
         int sh = RenderUtil.getScaledHeight();

         atheryx.render.scissor.Scissor.reset();
         Msdf.isReady();
         this.clampAllElements(sw, sh);
         boolean inEditorMode = this.isEditorActive();
         try { this.updateDynamicMetrics(); } catch (Throwable ignored) {}
         try { this.updateSessionCps(); } catch (Throwable ignored) {}

         // 1. Watermark
         try {
            if (this.getWatermarkSetting().isEnabled()) {
               this.renderWatermark(context, this.watermarkX, this.watermarkY);
            }
         } catch (Throwable ignored) {}

         // 2. Keybinds
         try {
            List<Function> activeBinds = this.getActiveKeyboundModules();
            boolean keybindsShouldShow = this.getKeybindsSetting().isEnabled() && (!activeBinds.isEmpty() || inEditorMode);
            this.keybindsOpenAnimation = this.animateTowards(this.keybindsOpenAnimation, keybindsShouldShow ? 1.0F : 0.0F, 0.14F);
            if (this.keybindsOpenAnimation > 0.01F) {
               this.renderKeybinds(context, this.keybindsX, this.keybindsY, activeBinds, this.keybindsOpenAnimation, inEditorMode);
            }
         } catch (Throwable ignored) {}

         // 3. Staff List
         try {
            List<StaffMember> staffList = this.collectOnlineStaff();
            boolean staffShouldShow = this.staffListBooleanSetting.isEnabled() && (!staffList.isEmpty() || inEditorMode);
            this.staffListOpenAnimation = this.animateTowards(this.staffListOpenAnimation, staffShouldShow ? 1.0F : 0.0F, 0.14F);
            if (this.staffListOpenAnimation > 0.01F) {
               this.renderStaffList(context, this.staffListX, this.staffListY, staffList, this.staffListOpenAnimation, inEditorMode);
            }
         } catch (Throwable ignored) {}

         // 4. Potions
         try {
            List<class_1293> activePotions = this.getActivePotionsList();
            boolean potionsShouldShow = this.getPotionsSetting().isEnabled() && (!activePotions.isEmpty() || inEditorMode);
            this.potionsOpenAnimation = this.animateTowards(this.potionsOpenAnimation, potionsShouldShow ? 1.0F : 0.0F, 0.12F);
            if (this.potionsOpenAnimation > 0.01F) {
               this.renderPotions(context, this.potionsX, this.potionsY, activePotions, this.potionsOpenAnimation, inEditorMode);
            }
         } catch (Throwable ignored) {}

         // 5. TargetHUD
         try {
            class_1309 auraTarget = this.findAuraTarget();
            if (auraTarget != null) {
               this.currentTarget = auraTarget;
            }
            boolean isChatOrEditor = inEditorMode || (this.mc.field_1755 instanceof class_408);
            boolean targetHudShouldShow = this.getTargetHudSetting().isEnabled() && (auraTarget != null || isChatOrEditor);
            this.targetHudOpenAnimation = this.animateTowards(this.targetHudOpenAnimation, targetHudShouldShow ? 1.0F : 0.0F, 0.15F);
            if (this.targetHudOpenAnimation > 0.01F) {
               class_1309 renderEntity = auraTarget != null ? auraTarget :
                                         (isChatOrEditor ? this.mc.field_1724 : this.currentTarget);
               if (renderEntity != null) {
                  this.renderTargetHud(context, this.targetHudX, this.targetHudY, renderEntity, this.targetHudOpenAnimation, isChatOrEditor);
               }
            } else {
               this.currentTarget = null;
            }
         } catch (Throwable ignored) {}


         // 10. Toast Notifications
         try {
            if (this.notificationsBooleanSetting.isEnabled()) {
               this.renderNotifications(context, (float)sw, (float)sh);
            }
         } catch (Throwable ignored) {}

         // 11. Inside Coordinates
         try {
            if (this.isInsideStyle()) {
               this.renderInsideCoordinates(context, sw, sh);
            }
         } catch (Throwable ignored) {}

         // 12. Interactive Editor Overlays (Guidelines, handles, context menu)
         try {
            if (inEditorMode) {
               this.renderEditorOverlays(context, sw, sh);
            } else {
               this.contextMenuOpen = false;
               this.contextMenuDragging = false;
            }
         } catch (Throwable ignored) {}
         atheryx.render.scissor.Scissor.reset();
      } catch (Throwable t) {
         // Prevent any render loop breakage
      }
   }

   // -------------------------------------------------------------------------
   // Dynamic Metrics & CPS Updating
   // -------------------------------------------------------------------------
   private void updateDynamicMetrics() {
      long now = System.currentTimeMillis();
      float dt = Math.min(0.1F, (float)(now - this.lastMetricsUpdateTime) * 0.001F);
      this.lastMetricsUpdateTime = now;

      float rawFps = (float)this.mc.method_47599();
      this.smoothedFps = this.lerp(this.smoothedFps, rawFps, dt * 8.0F);

      float rawPing = 0.0F;
      if (this.mc.method_1562() != null && this.mc.field_1724 != null) {
         class_640 playerEntry = this.mc.method_1562().method_2871(this.mc.field_1724.method_5667());
         if (playerEntry != null) {
            rawPing = (float)playerEntry.method_2959();
         }
      }
      this.smoothedPing = this.lerp(this.smoothedPing, rawPing, dt * 6.0F);
   }

   private void updateSessionCps() {
      try {
         if (this.mc == null || this.mc.method_22683() == null) {
            return;
         }
         long window = this.mc.method_22683().method_4490();
         if (window == 0L) {
            return;
         }
         boolean left = GLFW.glfwGetMouseButton(window, GLFW.GLFW_MOUSE_BUTTON_LEFT) == GLFW.GLFW_PRESS;
         boolean right = GLFW.glfwGetMouseButton(window, GLFW.GLFW_MOUSE_BUTTON_RIGHT) == GLFW.GLFW_PRESS;

         long now = System.currentTimeMillis();
         synchronized (leftClicks) {
            if (left && !lastLeftState) {
               leftClicks.add(now);
            }
            while (!leftClicks.isEmpty() && now - leftClicks.peek() > 1000L) {
               leftClicks.poll();
            }
         }
         synchronized (rightClicks) {
            if (right && !lastRightState) {
               rightClicks.add(now);
            }
            while (!rightClicks.isEmpty() && now - rightClicks.peek() > 1000L) {
               rightClicks.poll();
            }
         }
         lastLeftState = left;
         lastRightState = right;
      } catch (Throwable ignored) {}
   }

   public static int getLeftCps() {
      synchronized (leftClicks) {
         return leftClicks.size();
      }
   }

   public static int getRightCps() {
      synchronized (rightClicks) {
         return rightClicks.size();
      }
   }

   // -------------------------------------------------------------------------
   // Notification Toast Engine

   public static final class HudNotification {
      private final String title;
      private final String statusText;
      private final NotificationType type;
      private final long createdAt;
      private final long duration;
      private float slideProgress = 0.0F;
      private float alphaProgress = 0.0F;
      private float animatedY = -1.0F;
      private boolean dismissed = false;

      public HudNotification(String title, String statusText, NotificationType type, long duration) {
         this.title = title;
         this.statusText = statusText;
         this.type = type;
         this.duration = duration;
         this.createdAt = System.currentTimeMillis();
      }

      public String getTitle() { return this.title; }
      public String getStatusText() { return this.statusText; }
      public NotificationType getType() { return this.type; }
      public float getSlideProgress() { return this.slideProgress; }
      public void setSlideProgress(float val) { this.slideProgress = val; }
      public float getAlphaProgress() { return this.alphaProgress; }
      public void setAlphaProgress(float val) { this.alphaProgress = val; }
      public float getAnimatedY() { return this.animatedY; }
      public void setAnimatedY(float val) { this.animatedY = val; }

      public float getRemainingPercent() {
         long elapsed = System.currentTimeMillis() - this.createdAt;
         return Math.max(0.0F, Math.min(1.0F, 1.0F - (float)elapsed / (float)this.duration));
      }

      public boolean isExpired() {
         return this.dismissed || (System.currentTimeMillis() - this.createdAt >= this.duration);
      }
   }

   public static void onModuleToggle(String moduleName, boolean enabled) {
      String status = enabled ? "was enabled" : "was disabled";
      NotificationType type = enabled ? NotificationType.ENABLED : NotificationType.DISABLED;
      addNotification(moduleName, status, type);
   }

   public static void addNotification(String title, String message, NotificationType type) {
      if (Instance == null || !Instance.notificationsBooleanSetting.isEnabled()) {
         return;
      }
      HudNotification notification = new HudNotification(title, message, type, 2600L);
      synchronized (notificationQueue) {
         if (notificationQueue.size() >= 6) {
            notificationQueue.remove(0);
         }
         notificationQueue.add(notification);
      }
   }

   private void renderNotifications(class_332 context, float screenWidth, float screenHeight) {
      boolean inEditor = this.isEditorActive();
      if (notificationQueue.isEmpty() && !inEditor) {
         return;
      }

      float centerX = screenWidth / 2.0F;
      float crosshairY = screenHeight / 2.0F;
      float startY = crosshairY + this.notificationsYOffset;
      float toastGap = 3.5F;

      MsdfFont font = this.getInsideFont();
      MsdfFont boldFont = this.getInsideBoldFont();
      float fontSize = 6.8F;

      if (notificationQueue.isEmpty() && inEditor) {
         // Preview card in interactive editor
         String prefix = "Module ";
         String moduleName = "AttackAura";
         String middle = " was ";
         String statusWord = "enabled";

         float prefixW = Msdf.getTextWidth(font, prefix, fontSize);
         float nameW = Msdf.getTextWidth(boldFont, moduleName, fontSize);
         float middleW = Msdf.getTextWidth(font, middle, fontSize);
         float statusW = Msdf.getTextWidth(boldFont, statusWord, fontSize);
         float totalTextW = prefixW + nameW + middleW + statusW;

         float paddingH = 12.0F;
         float paddingV = 4.5F;
         float toastWidth = totalTextW + paddingH * 2.0F;
         float toastHeight = 16.5F;

         this.notificationsWidth = toastWidth;
         this.notificationsHeight = toastHeight;

         float toastX = centerX - toastWidth / 2.0F;
         this.renderSingleNotificationCard(context, null, NotificationType.ENABLED, toastX, startY, toastWidth, toastHeight, 0.95F,
            font, boldFont, fontSize, prefix, moduleName, middle, statusWord, prefixW, nameW, middleW, totalTextW);
         return;
      }

      synchronized (notificationQueue) {
         Iterator<HudNotification> it = notificationQueue.iterator();
         int index = 0;

         while (it.hasNext()) {
            HudNotification toast = it.next();
            boolean expired = toast.isExpired();

            float targetSlide = expired ? 0.0F : 1.0F;
            float targetAlpha = expired ? 0.0F : 1.0F;
            float slideSpeed = expired ? 0.16F : 0.14F;
            float alphaSpeed = expired ? 0.15F : 0.17F;

            toast.setSlideProgress(this.lerp(toast.getSlideProgress(), targetSlide, slideSpeed));
            toast.setAlphaProgress(this.lerp(toast.getAlphaProgress(), targetAlpha, alphaSpeed));

            if (expired && toast.getAlphaProgress() < 0.03F) {
               it.remove();
               continue;
            }

            float rawAlpha = Math.max(0.0F, Math.min(1.0F, toast.getAlphaProgress()));
            float easedAlpha = Easings.easeOutCubic(rawAlpha);

            float rawSlide = Math.max(0.0F, Math.min(1.0F, toast.getSlideProgress()));
            float easedSlide = expired ? Easings.easeOutCubic(rawSlide) : Easings.easeOutBack(rawSlide);

            // Measure text to determine toast width
            String prefix = "Module ";
            String moduleName = toast.getTitle();
            String middle = " has been ";
            String statusWord = toast.getType() == NotificationType.ENABLED ? "enabled" :
                                toast.getType() == NotificationType.DISABLED ? "disabled" : toast.getStatusText();

            float prefixW = Msdf.getTextWidth(font, prefix, fontSize);
            float nameW = Msdf.getTextWidth(boldFont, moduleName, fontSize);
            float middleW = Msdf.getTextWidth(font, middle, fontSize);
            float statusW = Msdf.getTextWidth(boldFont, statusWord, fontSize);
            float totalTextW = prefixW + nameW + middleW + statusW;

            float paddingH = 12.0F;
            float paddingV = 4.5F;
            float toastWidth = totalTextW + paddingH * 2.0F;
            float toastHeight = 16.5F;

            if (index == 0) {
               this.notificationsWidth = toastWidth;
               this.notificationsHeight = toastHeight;
            }

            // Smooth entrance slide (starts +13px lower and pops up)
            // Smooth exit slide (floats upwards by -9px while fading)
            float enterYOffset = (1.0F - easedSlide) * 13.0F;
            float exitYOffset = (1.0F - rawSlide) * -9.0F;
            float animYOffset = expired ? exitYOffset : enterYOffset;

            // Smooth micro-scale (0.92 -> 1.0 on enter, 1.0 -> 0.92 on exit)
            float scaleProgress = expired ? rawSlide : Math.min(1.0F, easedSlide);
            float scale = 0.92F + 0.08F * scaleProgress;

            float scaledWidth = toastWidth * scale;
            float scaledHeight = toastHeight * scale;

            // Target slot Y with smooth vertical stack interpolation
            float targetSlotY = startY + (float)index * (toastHeight + toastGap);
            if (toast.getAnimatedY() < 0.0F) {
               toast.setAnimatedY(targetSlotY);
            } else {
               toast.setAnimatedY(this.lerp(toast.getAnimatedY(), targetSlotY, 0.22F));
            }

            float finalToastY = toast.getAnimatedY() + animYOffset + (toastHeight - scaledHeight) / 2.0F;
            float finalToastX = centerX - scaledWidth / 2.0F;

            this.renderSingleNotificationCard(context, toast, toast.getType(), finalToastX, finalToastY, scaledWidth, scaledHeight, easedAlpha,
               font, boldFont, fontSize, prefix, moduleName, middle, statusWord, prefixW, nameW, middleW, totalTextW);
            index++;
         }
      }
   }

   private void renderSingleNotificationCard(class_332 context, HudNotification toast, NotificationType type,
         float x, float y, float width, float height, float alpha,
         MsdfFont font, MsdfFont boldFont, float fontSize,
         String prefix, String moduleName, String middle, String statusWord,
         float prefixW, float nameW, float middleW, float totalTextW) {

      float radius = 4.5F;
      this.drawInsidePanel(x, y, width, height, radius, alpha * 0.94F);

      // Text color
      Color textColor = this.withAlpha(new Color(200, 205, 218), alpha);
      Color nameColor = this.withAlpha(new Color(255, 255, 255), alpha);

      // Status color: green for enabled, red for disabled
      Color statusColor;
      if (type == NotificationType.ENABLED) {
         statusColor = this.withAlpha(new Color(46, 213, 115), alpha);
      } else if (type == NotificationType.DISABLED) {
         statusColor = this.withAlpha(new Color(255, 71, 87), alpha);
      } else {
         statusColor = this.withAlpha(ThemeManager.getCurrentColor(), alpha);
      }

      float textY = y + (height - fontSize) / 2.0F - 0.5F;

      cursorX += nameW;

      Msdf.drawText(context, font, middle, cursorX, textY, fontSize, textColor.getRGB(), false);
      cursorX += middleW;

      Msdf.drawText(context, boldFont, statusWord, cursorX, textY, fontSize, statusColor.getRGB(), false);
   }

   private void updateTargetHudSettingsPosition(int screenWidth, int screenHeight) {
      float menuWidth = 118.0F;
      float menuHeight = 72.0F;
      float gap = 6.0F;

      // Always position strictly to the right of TargetHUD without overlapping
      float targetX = this.targetHudX + this.targetHudWidth + gap;
      // If there is no space on the right edge of the screen, place to the left of TargetHUD
      if (targetX + menuWidth > (float)screenWidth - 4.0F) {
         targetX = this.targetHudX - menuWidth - gap;
      }
      if (targetX < 4.0F) {
         targetX = Math.max(4.0F, (float)screenWidth - menuWidth - 4.0F);
      }

      float targetY = Math.max(4.0F, Math.min(this.targetHudY, (float)screenHeight - menuHeight - 4.0F));

      this.contextMenuX = targetX;
      this.contextMenuY = targetY;
   }

   private void handleDragging(float mouseX, float mouseY, int screenWidth, int screenHeight) {
      long window = this.mc.method_22683().method_4490();
      boolean leftPressed = GLFW.glfwGetMouseButton(window, GLFW.GLFW_MOUSE_BUTTON_LEFT) == GLFW.GLFW_PRESS;
      boolean rightPressed = GLFW.glfwGetMouseButton(window, GLFW.GLFW_MOUSE_BUTTON_RIGHT) == GLFW.GLFW_PRESS;

      boolean leftJustPressed = leftPressed && !this.lastEditorLeftPressed;
      boolean rightJustPressed = rightPressed && !this.lastEditorRightPressed;
      this.lastEditorLeftPressed = leftPressed;
      this.lastEditorRightPressed = rightPressed;

      // Handle Context Menu dragging and clicks if it is open
      if (this.contextMenuOpen) {
         float menuWidth = 118.0F;
         float menuHeight = 72.0F;
         float x = this.contextMenuX;
         float y = this.contextMenuY;

         // Active dragging of context menu
         if (leftPressed && this.contextMenuDragging) {
            this.contextMenuX = Math.max(2.0F, Math.min(mouseX - this.contextMenuDragOffsetX, (float)screenWidth - menuWidth - 2.0F));
            this.contextMenuY = Math.max(2.0F, Math.min(mouseY - this.contextMenuDragOffsetY, (float)screenHeight - menuHeight - 2.0F));
            atheryx.manager.ConfigManager.markDirty();
         }

         if (!leftPressed && this.contextMenuDragging) {
            this.contextMenuDragging = false;
            atheryx.manager.ConfigManager.saveCurrent();
         }

         if (leftJustPressed) {
            if (this.isPointInside(mouseX, mouseY, x, y, menuWidth, menuHeight)) {
               // 1. Close button '×'
               if (this.isPointInside(mouseX, mouseY, x + menuWidth - 16.0F, y + 2.0F, 14.0F, 14.0F)) {
                  this.contextMenuOpen = false;
                  this.contextMenuDragging = false;
                  return;
               }

               // 2. Setting checkbox row: "Тряска при ударе"
               if (this.isPointInside(mouseX, mouseY, x + 4.0F, y + 18.0F, menuWidth - 8.0F, 16.0F)) {
                  this.targetHudHeadShakeBooleanSetting.setEnabled(!this.targetHudHeadShakeBooleanSetting.isEnabled());
                  atheryx.manager.ConfigManager.saveCurrent();
                  if (this.targetHudHeadShakeBooleanSetting.isEnabled()) {
                     this.triggerHeadShake();
                  }
                  return;
               }

               // 3. Option: "Сбросить позицию"
               if (this.isPointInside(mouseX, mouseY, x + 4.0F, y + 36.0F, menuWidth - 8.0F, 14.0F)) {
                  this.resetElementPosition(HudElement.TARGET_HUD);
                  this.contextMenuMoved = false;
                  this.updateTargetHudSettingsPosition(screenWidth, screenHeight);
                  this.contextMenuOpen = false;
                  this.contextMenuDragging = false;
                  atheryx.manager.ConfigManager.markDirty();
                  atheryx.manager.ConfigManager.saveCurrent();
                  return;
               }

               // 4. Option: "Закрыть"
               if (this.isPointInside(mouseX, mouseY, x + 4.0F, y + 52.0F, menuWidth - 8.0F, 14.0F)) {
                  this.contextMenuOpen = false;
                  this.contextMenuDragging = false;
                  return;
               }

               // Anywhere else on the settings card (header, padding) -> start dragging settings window!
               this.contextMenuDragging = true;
               this.contextMenuMoved = true;
               this.contextMenuDragOffsetX = mouseX - this.contextMenuX;
               this.contextMenuDragOffsetY = mouseY - this.contextMenuY;
               return;
            } else {
               // Click outside menu: check if clicking on TargetHUD
               DragTarget targetHovered = this.findHoveredDragTarget(mouseX, mouseY);
               if (targetHovered != DragTarget.TARGET_HUD && !this.contextMenuDragging) {
                  this.contextMenuOpen = false;
               }
            }
         }

         // Block interaction with HUD elements underneath the context menu
         if (this.contextMenuDragging || this.isPointInside(mouseX, mouseY, x, y, menuWidth, menuHeight)) {
            return;
         }
      }

      // Context menu toggle (right click ONLY on TargetHUD)
      if (rightJustPressed && this.currentDragTarget == DragTarget.NONE) {
         DragTarget target = this.findHoveredDragTarget(mouseX, mouseY);
         if (target == DragTarget.TARGET_HUD) {
            if (this.contextMenuOpen) {
               this.contextMenuOpen = false;
               this.contextMenuDragging = false;
            } else {
               this.openTargetHudSettings(screenWidth, screenHeight);
            }
            return;
         } else if (this.contextMenuOpen) {
            this.contextMenuOpen = false;
            this.contextMenuDragging = false;
         }
      }

      // Mouse button down: start dragging
      if (leftPressed && !this.isDragging && this.currentDragTarget == DragTarget.NONE) {
         DragTarget target = this.findHoveredDragTarget(mouseX, mouseY);
         if (target != DragTarget.NONE) {
            if (target == DragTarget.TARGET_HUD) {
               // Check if clicking dummy head avatar in editor mode to test head shake
               if (leftJustPressed) {
                  float baseHeadX = this.targetHudX + 5.0F;
                  float baseHeadY = this.targetHudY + (44.0F - 34.0F) / 2.0F;
                  if (this.isPointInside(mouseX, mouseY, baseHeadX, baseHeadY, 34.0F, 34.0F)) {
                     this.triggerHeadShake();
                  }
               }
            } else {
               this.contextMenuOpen = false;
               this.contextMenuDragging = false;
            }

            this.currentDragTarget = target;
            this.isDragging = true;
            float[] pos = this.getElementPos(target);
            this.dragOffsetX = mouseX - pos[0];
            this.dragOffsetY = mouseY - pos[1];
         }
      }

      // Dragging active
      if (leftPressed && this.isDragging && this.currentDragTarget != DragTarget.NONE) {
         float newX = mouseX - this.dragOffsetX;
         float newY = mouseY - this.dragOffsetY;

         // Magnetic Snapping
         if (this.magneticSnappingBooleanSetting.isEnabled()) {
            float snapCenterX = (float)screenWidth / 2.0F;
            float snapCenterY = (float)screenHeight / 2.0F;

            float[] bounds = this.getElementBounds(this.currentDragTarget);
            float width = bounds[0];
            float height = bounds[1];

            if (this.currentDragTarget == DragTarget.NOTIFICATIONS) {
               newX = snapCenterX - width / 2.0F;
               this.activeSnapLineX = snapCenterX;
            } else if (Math.abs(newX + width / 2.0F - snapCenterX) < SNAP_DISTANCE) {
               newX = snapCenterX - width / 2.0F;
               this.activeSnapLineX = snapCenterX;
            } else if (Math.abs(newX - 8.0F) < SNAP_DISTANCE) {
               newX = 8.0F;
               this.activeSnapLineX = 8.0F;
            } else if (Math.abs(newX + width - ((float)screenWidth - 8.0F)) < SNAP_DISTANCE) {
               newX = (float)screenWidth - 8.0F - width;
               this.activeSnapLineX = (float)screenWidth - 8.0F;
            } else {
               this.activeSnapLineX = -1.0F;
            }

            if (Math.abs(newY + height / 2.0F - snapCenterY) < SNAP_DISTANCE) {
               newY = snapCenterY - height / 2.0F;
               this.activeSnapLineY = snapCenterY;
            } else if (Math.abs(newY - 8.0F) < SNAP_DISTANCE) {
               newY = 8.0F;
               this.activeSnapLineY = 8.0F;
            } else if (Math.abs(newY + height - ((float)screenHeight - 8.0F)) < SNAP_DISTANCE) {
               newY = (float)screenHeight - 8.0F - height;
               this.activeSnapLineY = (float)screenHeight - 8.0F;
            } else {
               this.activeSnapLineY = -1.0F;
            }
         }

         this.setElementPos(this.currentDragTarget, newX, newY, screenWidth, screenHeight);

         if (this.currentDragTarget == DragTarget.TARGET_HUD && this.contextMenuOpen && !this.contextMenuMoved) {
            this.updateTargetHudSettingsPosition(screenWidth, screenHeight);
         }
      }

      // Drag release
      if (!leftPressed && this.isDragging) {
         this.isDragging = false;
         this.currentDragTarget = DragTarget.NONE;
         this.activeSnapLineX = -1.0F;
         this.activeSnapLineY = -1.0F;
         atheryx.manager.ConfigManager.markDirty();
         atheryx.manager.ConfigManager.saveCurrent();
      }
   }

   private DragTarget findHoveredDragTarget(float mouseX, float mouseY) {
      DragTarget[] targets = new DragTarget[]{
         DragTarget.WATERMARK,
         DragTarget.KEYBINDS,
         DragTarget.STAFF_LIST,
         DragTarget.POTIONS,
         DragTarget.TARGET_HUD,
         DragTarget.SCOREBOARD,
         DragTarget.NEURO_AIM,
         DragTarget.SESSION_INFO,
         DragTarget.NOTIFICATIONS
      };

      for (DragTarget target : targets) {
         float[] pos = this.getElementPos(target);
         float[] bounds = this.getElementBounds(target);
         if (this.isPointInside(mouseX, mouseY, pos[0], pos[1], bounds[0], bounds[1])) {
            return target;
         }
      }
      return DragTarget.NONE;
   }

   public boolean isPointInside(float x, float y, float rx, float ry, float rw, float rh) {
      return x >= rx && x <= rx + rw && y >= ry && y <= ry + rh;
   }

   }

   private Color lerpColor(Color from, Color to, float progress) {
      float p = Math.max(0.0F, Math.min(1.0F, progress));
      int r = (int)((float)from.getRed() + (float)(to.getRed() - from.getRed()) * p);
      int g = (int)((float)from.getGreen() + (float)(to.getGreen() - from.getGreen()) * p);
      int b = (int)((float)from.getBlue() + (float)(to.getBlue() - from.getBlue()) * p);
      int a = (int)((float)from.getAlpha() + (float)(to.getAlpha() - from.getAlpha()) * p);
      return new Color(r, g, b, a);
   }

   private float[] getElementPos(DragTarget target) {
      return switch (target) {
         case WATERMARK -> new float[]{this.watermarkX, this.watermarkY};
         case KEYBINDS -> new float[]{this.keybindsX, this.keybindsY};
         case STAFF_LIST -> new float[]{this.staffListX, this.staffListY};
         case POTIONS -> new float[]{this.potionsX, this.potionsY};
         case TARGET_HUD -> new float[]{this.targetHudX, this.targetHudY};

   private String cleanStaffTag(String rawPrefix, boolean vanished) {
      if (vanished) return "VANISH";
      if (rawPrefix == null || rawPrefix.isEmpty()) return "";

      // 1. Strip Minecraft formatting codes (§a, §b, etc.)
      String clean = rawPrefix.replaceAll("(?i)§[0-9a-fk-or]", "");
      // 2. Remove brackets, punctuation, and extra whitespace
      clean = clean.replace("[", "").replace("]", "").replace("(", "").replace(")", "")
                   .replace(":", "").replace("»", "").replace(">", "").replace("|", "").trim();

      // 3. Keep only clean letters and digits
      StringBuilder sb = new StringBuilder();
      for (char c : clean.toCharArray()) {
         if (Character.isLetterOrDigit(c) || c == '_' || c == '-') {
            sb.append(c);
         }
      }
      clean = sb.toString().trim();

      if (clean.isEmpty()) return "";
      return clean.toUpperCase(Locale.ROOT);
   }

   private String sanitizeScoreboardText(String text) {
      if (text == null || text.isEmpty()) return "";
      text = text.replaceAll("(?i)(?:§[0-9a-fk-or]|\\?[0-9a-fk-or])+\\s*$", "");
      StringBuilder sb = new StringBuilder();
      for (int i = 0; i < text.length(); i++) {
         char c = text.charAt(i);
         if (c < 32 && c != '\t') continue;
         if (c == 0x200B || c == 0xFEFF || (c >= 0xE000 && c <= 0xF8FF)) continue;
         if (c == '⬝' || c == '▪' || c == '✦' || c == '•' || c == '●' || c == '◆' || c == '❖' || c == '★' || c == '⭐' || c == '⚡' || c == '⛂' || c == '⛃' || c == '✪') {
            continue;
         }
         if (c == '–' || c == '—') {
            sb.append('-');
            continue;
         }
         if (c == '…') {
            sb.append("...");
            continue;
         }
         if (c == '»' || c == '›' || c == '▶' || c == '►') {
            sb.append('>');
            continue;
         }
         if (c == '«' || c == '‹' || c == '◀' || c == '◄') {
            sb.append('<');
            continue;
         }
         if (c >= 32 && c <= 126) {
            sb.append(c);
         } else if (c == 1025 || (c >= 1040 && c <= 1103) || c == 1105) {
            sb.append(c);
         }
      }
      return sb.toString();
   }

   public Color resolveColor(Color c1, Color c2, float ratio) {
      float r = this.clamp(ratio, 0.0F, 1.0F);
      float ir = 1.0F - r;
      int red = Math.round((float)c1.getRed() * ir + (float)c2.getRed() * r);
      int green = Math.round((float)c1.getGreen() * ir + (float)c2.getGreen() * r);
      int blue = Math.round((float)c1.getBlue() * ir + (float)c2.getBlue() * r);
      int alpha = Math.round((float)c1.getAlpha() * ir + (float)c2.getAlpha() * r);
      return new Color(this.clamp(red, 0, 255), this.clamp(green, 0, 255), this.clamp(blue, 0, 255), this.clamp(alpha, 0, 255));
   }

   }

   public String truncateText(String text, float maxWidth, MsdfFont font, float fontSize) {
      if (text == null) return "";
      float fullWidth = Msdf.getTextWidth(font, text, fontSize);
      if (fullWidth <= maxWidth) {
         return text;
      }
      String ellipsis = "...";
      float ellW = Msdf.getTextWidth(font, ellipsis, fontSize);
      StringBuilder sb = new StringBuilder();
      for (int i = 0; i < text.length(); i++) {
         char c = text.charAt(i);
         if (Msdf.getTextWidth(font, sb.toString() + c, fontSize) + ellW > maxWidth) {
            break;
         }
         sb.append(c);
      }
      return sb.toString() + ellipsis;
   }

         case 6 -> "VI";
         case 7 -> "VII";
         case 8 -> "VIII";
         case 9 -> "IX";
         case 10 -> "X";
         default -> String.valueOf(number);
      };
   }

         float r = Math.max(0.0F, Math.min(1.0F, ratio));
         float ir = 1.0F - r;
         int red = Math.round((float)c1.getRed() * ir + (float)c2.getRed() * r);
         int green = Math.round((float)c1.getGreen() * ir + (float)c2.getGreen() * r);
         int blue = Math.round((float)c1.getBlue() * ir + (float)c2.getBlue() * r);
         int alpha = Math.round((float)c1.getAlpha() * ir + (float)c2.getAlpha() * r);
         return new Color(
            Math.max(0, Math.min(255, red)),
            Math.max(0, Math.min(255, green)),
            Math.max(0, Math.min(255, blue)),
            Math.max(0, Math.min(255, alpha))
         );
      }

      public static Color darken(Color color, float factor) {
         float f = Math.max(0.0F, Math.min(1.0F, 1.0F - factor));
         return new Color(
            Math.max(0, Math.min(255, Math.round((float)color.getRed() * f))),
            Math.max(0, Math.min(255, Math.round((float)color.getGreen() * f))),
            Math.max(0, Math.min(255, Math.round((float)color.getBlue() * f))),
            color.getAlpha()
         );
      }

      public static Color brighten(Color color, float factor) {
         float f = Math.max(0.0F, Math.min(1.0F, factor));
         int r = color.getRed() + Math.round((float)(255 - color.getRed()) * f);
         int g = color.getGreen() + Math.round((float)(255 - color.getGreen()) * f);
         int b = color.getBlue() + Math.round((float)(255 - color.getBlue()) * f);
         return new Color(
            Math.max(0, Math.min(255, r)),
            Math.max(0, Math.min(255, g)),
            Math.max(0, Math.min(255, b)),
            color.getAlpha()
         );
      }

      public static Color pulseColor(Color base, Color target, float speed) {
         float wave = (float)((Math.sin((double)System.currentTimeMillis() * 0.001D * (double)speed) + 1.0D) * 0.5D);
         return mixColors(base, target, wave);
      }

      public static Color cycleHue(float saturation, float brightness, float speed) {
         float hue = (float)((System.currentTimeMillis() % (long)(1000.0F / speed)) / (1000.0F / speed));
         return Color.getHSBColor(hue, Math.max(0.0F, Math.min(1.0F, saturation)), Math.max(0.0F, Math.min(1.0F, brightness)));
      }

      public static Color fromHex(String hex) {
         if (hex == null || hex.isEmpty()) {
            return Color.WHITE;
         }
         String clean = hex.startsWith("#") ? hex.substring(1) : hex;
         try {
            if (clean.length() == 6) {
               int rgb = Integer.parseInt(clean, 16);
               return new Color(rgb);
            } else if (clean.length() == 8) {
               int argb = (int)Long.parseLong(clean, 16);
               return new Color(argb, true);
            }
         } catch (NumberFormatException ignored) {}
         return Color.WHITE;
      }

      public static String toHex(Color color) {
         return String.format("#%02X%02X%02X", color.getRed(), color.getGreen(), color.getBlue());
      }

      public static String toHexWithAlpha(Color color) {
         return String.format("#%02X%02X%02X%02X", color.getAlpha(), color.getRed(), color.getGreen(), color.getBlue());
      }

      public static double getLuminance(Color color) {
         double r = (double)color.getRed() / 255.0D;
         double g = (double)color.getGreen() / 255.0D;
         double b = (double)color.getBlue() / 255.0D;
         return 0.2126D * r + 0.7152D * g + 0.0722D * b;
      }

      public static boolean isDark(Color color) {
         return getLuminance(color) < 0.5D;
      }

      public static Color getContrastColor(Color background) {
         return isDark(background) ? Color.WHITE : new Color(20, 20, 25);
      }

      public static Color getComplementary(Color color) {
         return new Color(255 - color.getRed(), 255 - color.getGreen(), 255 - color.getBlue(), color.getAlpha());
      }

      public static Color[] createGradientStops(Color start, Color end, int steps) {
         int safeSteps = Math.max(2, steps);
         Color[] result = new Color[safeSteps];
         for (int i = 0; i < safeSteps; i++) {
            float ratio = (float)i / (float)(safeSteps - 1);
            result[i] = mixColors(start, end, ratio);
         }
         return result;
      }
   }

   /**

   public static final class GuiAnimationMath {
      private GuiAnimationMath() {}

      public static float clamp01(float val) {
         return Math.max(0.0F, Math.min(1.0F, val));
      }

      public static float lerp(float start, float end, float pct) {
         return start + clamp01(pct) * (end - start);
      }

      public static double lerp(double start, double end, double pct) {
         double p = Math.max(0.0D, Math.min(1.0D, pct));
         return start + p * (end - start);
      }

      public static float easeInSine(float x) {
         float t = clamp01(x);
         return 1.0F - (float)Math.cos((double)t * Math.PI / 2.0D);
      }

      public static float easeOutSine(float x) {
         float t = clamp01(x);
         return (float)Math.sin((double)t * Math.PI / 2.0D);
      }

      public static float easeInOutSine(float x) {
         float t = clamp01(x);
         return -((float)Math.cos(Math.PI * (double)t) - 1.0F) / 2.0F;
      }

      public static float easeInQuad(float x) {
         float t = clamp01(x);
         return t * t;
      }

      public static float easeOutQuad(float x) {
         float t = clamp01(x);
         return 1.0F - (1.0F - t) * (1.0F - t);
      }

      public static float easeInOutQuad(float x) {
         float t = clamp01(x);
         return t < 0.5F ? 2.0F * t * t : 1.0F - (float)Math.pow(-2.0F * t + 2.0F, 2.0D) / 2.0F;
      }

      public static float easeInCubic(float x) {
         float t = clamp01(x);
         return t * t * t;
      }

      public static float easeOutCubic(float x) {
         float t = clamp01(x);
         return 1.0F - (float)Math.pow(1.0F - t, 3.0D);
      }

      public static float easeInOutCubic(float x) {
         float t = clamp01(x);
         return t < 0.5F ? 4.0F * t * t * t : 1.0F - (float)Math.pow(-2.0F * t + 2.0F, 3.0D) / 2.0F;
      }

      public static float easeInQuart(float x) {
         float t = clamp01(x);
         return t * t * t * t;
      }

      public static float easeOutQuart(float x) {
         float t = clamp01(x);
         return 1.0F - (float)Math.pow(1.0F - t, 4.0D);
      }

      public static float easeInOutQuart(float x) {
         float t = clamp01(x);
         return t < 0.5F ? 8.0F * t * t * t * t : 1.0F - (float)Math.pow(-2.0F * t + 2.0F, 4.0D) / 2.0F;
      }

      public static float easeInQuint(float x) {
         float t = clamp01(x);
         return t * t * t * t * t;
      }

      public static float easeOutQuint(float x) {
         float t = clamp01(x);
         return 1.0F - (float)Math.pow(1.0F - t, 5.0D);
      }

      public static float easeInOutQuint(float x) {
         float t = clamp01(x);
         return t < 0.5F ? 16.0F * t * t * t * t * t : 1.0F - (float)Math.pow(-2.0F * t + 2.0F, 5.0D) / 2.0F;
      }

      public static float easeInExpo(float x) {
         float t = clamp01(x);
         return t == 0.0F ? 0.0F : (float)Math.pow(2.0D, 10.0D * (double)(t - 1.0F));
      }

      public static float easeOutExpo(float x) {
         float t = clamp01(x);
         return t == 1.0F ? 1.0F : 1.0F - (float)Math.pow(2.0D, -10.0D * (double)t);
      }

      public static float easeInOutExpo(float x) {
         float t = clamp01(x);
         if (t == 0.0F) return 0.0F;
         if (t == 1.0F) return 1.0F;
         return t < 0.5F ?
            (float)Math.pow(2.0D, 20.0D * (double)t - 10.0D) / 2.0F :
            (2.0F - (float)Math.pow(2.0D, -20.0D * (double)t + 10.0D)) / 2.0F;
      }

      public static float easeInCirc(float x) {
         float t = clamp01(x);
         return 1.0F - (float)Math.sqrt(1.0D - Math.pow(t, 2.0D));
      }

      public static float easeOutCirc(float x) {
         float t = clamp01(x);
         return (float)Math.sqrt(1.0D - Math.pow(t - 1.0F, 2.0D));
      }

      public static float easeInOutCirc(float x) {
         float t = clamp01(x);
         return t < 0.5F ?
            (1.0F - (float)Math.sqrt(1.0D - Math.pow(2.0F * t, 2.0D))) / 2.0F :
            ((float)Math.sqrt(1.0D - Math.pow(-2.0F * t + 2.0F, 2.0D)) + 1.0F) / 2.0F;
      }

      public static float easeInBack(float x) {
         float t = clamp01(x);
         float c1 = 1.70158F;
         float c3 = c1 + 1.0F;
         return c3 * t * t * t - c1 * t * t;
      }

      public static float easeOutBack(float x) {
         float t = clamp01(x);
         float c1 = 1.70158F;
         float c3 = c1 + 1.0F;
         return 1.0F + c3 * (float)Math.pow(t - 1.0F, 3.0D) + c1 * (float)Math.pow(t - 1.0F, 2.0D);
      }

      public static float easeInOutBack(float x) {
         float t = clamp01(x);
         float c1 = 1.70158F;
         float c2 = c1 * 1.525F;
         return t < 0.5F ?
            ((float)Math.pow(2.0F * t, 2.0D) * ((c2 + 1.0F) * 2.0F * t - c2)) / 2.0F :
            ((float)Math.pow(2.0F * t - 2.0F, 2.0D) * ((c2 + 1.0F) * (t * 2.0F - 2.0F) + c2) + 2.0F) / 2.0F;
      }

      public static float easeOutBounce(float x) {
         float t = clamp01(x);
         float n1 = 7.5625F;
         float d1 = 2.75F;
         if (t < 1.0F / d1) {
            return n1 * t * t;
         } else if (t < 2.0F / d1) {
            t -= 1.5F / d1;
            return n1 * t * t + 0.75F;
         } else if (t < 2.5F / d1) {
            t -= 2.25F / d1;
            return n1 * t * t + 0.9375F;
         } else {
            t -= 2.625F / d1;
            return n1 * t * t + 0.984375F;
         }
      }

      public static float easeInBounce(float x) {
         return 1.0F - easeOutBounce(1.0F - x);
      }

      public static float easeInOutBounce(float x) {
         float t = clamp01(x);
         return t < 0.5F ?
            (1.0F - easeOutBounce(1.0F - 2.0F * t)) / 2.0F :
            (1.0F + easeOutBounce(2.0F * t - 1.0F)) / 2.0F;
      }

      public static float easeOutElastic(float x) {
         float t = clamp01(x);
         if (t == 0.0F) return 0.0F;
         if (t == 1.0F) return 1.0F;
         float c4 = (float)(2.0D * Math.PI) / 3.0F;
         return (float)Math.pow(2.0D, -10.0D * (double)t) * (float)Math.sin(((double)t * 10.0D - 0.75D) * (double)c4) + 1.0F;
      }

      public static float easeInElastic(float x) {
         float t = clamp01(x);
         if (t == 0.0F) return 0.0F;
         if (t == 1.0F) return 1.0F;
         float c4 = (float)(2.0D * Math.PI) / 3.0F;
         return -(float)Math.pow(2.0D, 10.0D * (double)t - 10.0D) * (float)Math.sin(((double)t * 10.0D - 10.75D) * (double)c4);
      }
   }

            case 6 -> "VI";
            case 7 -> "VII";

   // -------------------------------------------------------------------------
   // Interactive Magnetic Snapping & Alignment Guide Engine
   // -------------------------------------------------------------------------
   public static final class HudSnapEngine {
      public enum SnapTarget {
         SCREEN_LEFT, SCREEN_CENTER, SCREEN_RIGHT,
         SCREEN_TOP, SCREEN_MIDDLE, SCREEN_BOTTOM,
         OTHER_EDGE, OTHER_CENTER
      }

      public static final class SnapResult {
         public final boolean snappedX;
         public final boolean snappedY;
         public final float newX;
         public final float newY;
         public final float guideX;
         public final float guideY;
         public final SnapTarget targetX;
         public final SnapTarget targetY;

         public SnapResult(boolean sx, boolean sy, float nx, float ny, float gx, float gy, SnapTarget tx, SnapTarget ty) {
            this.snappedX = sx;
            this.snappedY = sy;
            this.newX = nx;
            this.newY = ny;
            this.guideX = gx;
            this.guideY = gy;
            this.targetX = tx;
            this.targetY = ty;
         }
      }

      private static float activeGuideX = -1.0F;
      private static float activeGuideY = -1.0F;
      private static float guideAlphaX = 0.0F;
      private static float guideAlphaY = 0.0F;
      private static final float SNAP_THRESHOLD = 7.0F;

      private HudSnapEngine() {}

      public static SnapResult computeSnap(HudElement dragging, float candidateX, float candidateY, float width, float height, float screenW, float screenH) {
         float resultX = candidateX;
         float resultY = candidateY;
         boolean snapX = false;
         boolean snapY = false;
         float gx = -1.0F;
         float gy = -1.0F;
         SnapTarget tx = null;
         SnapTarget ty = null;

         // Screen edge horizontal snapping
         if (Math.abs(candidateX) < SNAP_THRESHOLD) {
            resultX = 0.0F;
            snapX = true;
            gx = 0.0F;
            tx = SnapTarget.SCREEN_LEFT;
         } else if (Math.abs((candidateX + width * 0.5F) - screenW * 0.5F) < SNAP_THRESHOLD) {
            resultX = screenW * 0.5F - width * 0.5F;
            snapX = true;
            gx = screenW * 0.5F;
            tx = SnapTarget.SCREEN_CENTER;
         } else if (Math.abs((candidateX + width) - screenW) < SNAP_THRESHOLD) {
            resultX = screenW - width;
            snapX = true;
            gx = screenW;
            tx = SnapTarget.SCREEN_RIGHT;
         }

         // Screen edge vertical snapping
         if (Math.abs(candidateY) < SNAP_THRESHOLD) {
            resultY = 0.0F;
            snapY = true;
            gy = 0.0F;
            ty = SnapTarget.SCREEN_TOP;
         } else if (Math.abs((candidateY + height * 0.5F) - screenH * 0.5F) < SNAP_THRESHOLD) {
            resultY = screenH * 0.5F - height * 0.5F;
            snapY = true;
            gy = screenH * 0.5F;
            ty = SnapTarget.SCREEN_MIDDLE;
         } else if (Math.abs((candidateY + height) - screenH) < SNAP_THRESHOLD) {
            resultY = screenH - height;
            snapY = true;
            gy = screenH;
            ty = SnapTarget.SCREEN_BOTTOM;
         }

         // Module-to-module proximity magnetic snapping
         if (Instance != null) {
            for (HudElement other : HudElement.values()) {
               if (other == dragging) continue;
               float ox = Instance.getElementX(other);
               float oy = Instance.getElementY(other);
               float ow = Instance.getElementWidth(other);
               float oh = Instance.getElementHeight(other);

               // Align left with left
               if (!snapX && Math.abs(candidateX - ox) < SNAP_THRESHOLD) {
                  resultX = ox;
                  snapX = true;
                  gx = ox;
                  tx = SnapTarget.OTHER_EDGE;
               }
               // Align right with right
               if (!snapX && Math.abs((candidateX + width) - (ox + ow)) < SNAP_THRESHOLD) {
                  resultX = ox + ow - width;
                  snapX = true;
                  gx = ox + ow;
                  tx = SnapTarget.OTHER_EDGE;
               }
               // Align top with top
               if (!snapY && Math.abs(candidateY - oy) < SNAP_THRESHOLD) {
                  resultY = oy;
                  snapY = true;
                  gy = oy;
                  ty = SnapTarget.OTHER_EDGE;
               }
               // Align bottom with bottom
               if (!snapY && Math.abs((candidateY + height) - (oy + oh)) < SNAP_THRESHOLD) {
                  resultY = oy + oh - height;
                  snapY = true;
                  gy = oy + oh;
                  ty = SnapTarget.OTHER_EDGE;
               }
            }
         }

         if (snapX) {
            activeGuideX = gx;
            guideAlphaX = Math.min(1.0F, guideAlphaX + 0.15F);
         } else {
            guideAlphaX = Math.max(0.0F, guideAlphaX - 0.1F);
         }

         if (snapY) {
            activeGuideY = gy;
            guideAlphaY = Math.min(1.0F, guideAlphaY + 0.15F);
         } else {
            guideAlphaY = Math.max(0.0F, guideAlphaY - 0.1F);
         }

         return new SnapResult(snapX, snapY, resultX, resultY, gx, gy, tx, ty);
      }

   public static final class SessionTrackerEngine {
      private static final long sessionStartTime = System.currentTimeMillis();
      private static int sessionKills = 0;
      private static int sessionDeaths = 0;
      private static int currentKillstreak = 0;
      private static int highestKillstreak = 0;
      private static float damageDealt = 0.0F;
      private static float damageTaken = 0.0F;

      private static final Queue<Long> leftClicks = new LinkedList<>();
      private static final Queue<Long> rightClicks = new LinkedList<>();

      private SessionTrackerEngine() {}

      public static void onLeftClick() {
         synchronized (leftClicks) {
            leftClicks.add(System.currentTimeMillis());
         }
      }

      public static void onRightClick() {
         synchronized (rightClicks) {
            rightClicks.add(System.currentTimeMillis());
         }
      }

      public static int getLeftCps() {
         long now = System.currentTimeMillis();
         synchronized (leftClicks) {
            while (!leftClicks.isEmpty() && now - leftClicks.peek() > 1000L) {
               leftClicks.poll();
            }
            return leftClicks.size();
         }
      }

      public static int getRightCps() {
         long now = System.currentTimeMillis();
         synchronized (rightClicks) {
            while (!rightClicks.isEmpty() && now - rightClicks.peek() > 1000L) {
               rightClicks.poll();
            }
            return rightClicks.size();
         }
      }

      public static void onKill() {
         sessionKills++;
         currentKillstreak++;
         if (currentKillstreak > highestKillstreak) {
            highestKillstreak = currentKillstreak;
         }
      }

      public static void onDeath() {
         sessionDeaths++;
         currentKillstreak = 0;
      }

      public static void addDamageDealt(float amount) {
         damageDealt += Math.max(0.0F, amount);
      }

      public static void addDamageTaken(float amount) {
         damageTaken += Math.max(0.0F, amount);
      }

      public static float getKdr() {
         if (sessionDeaths == 0) return (float)sessionKills;
         return (float)sessionKills / (float)sessionDeaths;
      }

      public static String getSessionDurationFormatted() {
         long elapsed = System.currentTimeMillis() - sessionStartTime;
         long seconds = elapsed / 1000L;
         long s = seconds % 60L;
         long m = (seconds / 60L) % 60L;
         long h = seconds / 3600L;
         if (h > 0) {
            return String.format(Locale.US, "%02d:%02d:%02d", h, m, s);
         }
         return String.format(Locale.US, "%02d:%02d", m, s);
      }

            }
         }

         targetTooltipAlpha = activeTooltip != null ? 1.0F : 0.0F;
         tooltipAlpha += (targetTooltipAlpha - tooltipAlpha) * 0.2F;
      }

      public static void renderTooltip(class_332 context, double mouseX, double mouseY) {
         if (tooltipAlpha <= 0.02F || activeTooltip == null) return;

         MsdfFont font = Msdf.SF_REGULAR;
         if (font == null) return;

         float textW = Msdf.getTextWidth(font, activeTooltip, 6.5F);
         float padX = 6.0F;
         float padY = 4.0F;
         float cardW = textW + padX * 2.0F;
         float cardH = 14.0F;
         float tx = (float)mouseX + 8.0F;
         float ty = (float)mouseY - 18.0F;

         int alpha = Math.round(tooltipAlpha * 230.0F);
         Color bg = new Color(16, 16, 24, alpha);
         Color border = new Color(145, 110, 255, Math.round(tooltipAlpha * 180.0F));
         Color textColor = new Color(245, 245, 255, alpha);

         RenderUtil.drawRoundedRect(tx, ty, cardW, cardH, 4.0F, bg);
         RenderUtil.drawRoundedOutline(tx, ty, cardW, cardH, 4.0F, 1.0F, border);
         Msdf.drawText(context, font, activeTooltip, tx + padX, ty + padY, 6.5F, textColor.getRGB(), false);
      }
   }

   // -------------------------------------------------------------------------
   // Diagnostic Telemetry & Pipeline Benchmark Monitor
   // -------------------------------------------------------------------------
   public static final class HudDiagnosticTelemetry {
      private static final long[] frameDurations = new long[60];
      private static int sampleIndex = 0;
      private static long lastSampleTimestamp = System.currentTimeMillis();

      private HudDiagnosticTelemetry() {}

      public static void recordFrame(long durationNano) {
         frameDurations[sampleIndex % frameDurations.length] = durationNano;
         sampleIndex++;
         lastSampleTimestamp = System.currentTimeMillis();
      }

      public static float getAverageFrameDurationMs() {
         long total = 0L;
         int count = Math.min(sampleIndex, frameDurations.length);
         if (count == 0) return 0.0F;
         for (int i = 0; i < count; i++) {
            total += frameDurations[i];
         }
         return (float)total / (float)count / 1000000.0F;
      }

      public static void renderMiniGraph(class_332 context, float x, float y, float width, float height) {
         RenderUtil.drawRoundedRect(x, y, width, height, 4.0F, new Color(14, 14, 20, 200));
         int count = Math.min(sampleIndex, frameDurations.length);
         if (count < 2) return;

         float stepX = width / (float)frameDurations.length;
         for (int i = 0; i < count; i++) {
            float ms = (float)frameDurations[i] / 1000000.0F;
            float barH = Math.min(height, (ms / 16.67F) * height * 0.5F);
            Color barCol = ms > 16.67F ? new Color(255, 75, 85, 200) : new Color(100, 108, 125, 200);
            RenderUtil.drawRoundedRect(x + (float)i * stepX, y + height - barH, stepX, barH, 0.0F, barCol);
         }
      }
   }
